第 12 章：实战——SQLite + 文件系统 MCP Server

12.1 实战目标

本章把前面所有知识整合成一个可实际使用的 MCP Server，它提供两类能力：

  • SQLite 数据库：查询表结构、执行只读 SQL、插入记录。
  • 本地文件系统：在授权目录内读取文件、列出目录、写入文件。

12.3 SQLite Tools 实现

# db.py
import sqlite3
from pathlib import Path
from typing import Any
from mcp.server.fastmcp import FastMCP, Context

DB_PATH = Path("data.db")


def get_conn() -> sqlite3.Connection:
    conn = sqlite3.connect(DB_PATH)
    conn.row_factory = sqlite3.Row
    return conn


def register_db(mcp: FastMCP):

    @mcp.tool()
    def list_tables() -> list[str]:
        """列出数据库中所有表的名称。"""
        with get_conn() as conn:
            rows = conn.execute(
                "SELECT name FROM sqlite_master WHERE type='table' ORDER BY name"
            ).fetchall()
        return [r["name"] for r in rows]

    @mcp.tool()
    def describe_table(table: str) -> str:
        """返回指定表的列定义（列名、类型、是否非空、默认值）。"""
        # 防御 SQL 注入：表名不走参数化，先校验
        with get_conn() as conn:
            valid = [r[0] for r in conn.execute(
                "SELECT name FROM sqlite_master WHERE type='table'"
            ).fetchall()]
        if table not in valid:
            raise ValueError(f"表 '{table}' 不存在。")
        with get_conn() as conn:
            rows = conn.execute(f"PRAGMA table_info({table})").fetchall()
        lines = [
            f"{r['name']} {r['type']}"
            + (" NOT NULL" if r["notnull"] else "")
            + (f" DEFAULT {r['dflt_value']}" if r["dflt_value"] else "")
            for r in rows
        ]
        return "\n".join(lines)

    @mcp.tool()
    def query(sql: str, params: list[Any] | None = None) -> str:
        """执行只读 SQL 查询，返回结果的 Markdown 表格（最多 200 行）。
        注意：只允许 SELECT 语句，写操作请用 execute_write。
        """
        sql_strip = sql.strip().upper()
        if not sql_strip.startswith("SELECT"):
            raise ValueError("query 仅支持 SELECT 语句，写操作请用 execute_write。")
        with get_conn() as conn:
            rows = conn.execute(sql, params or []).fetchmany(200)
        if not rows:
            return "查询结果为空。"
        headers = rows[0].keys()
        lines = ["| " + " | ".join(headers) + " |",
                 "| " + " | ".join("---" for _ in headers) + " |"]
        for r in rows:
            lines.append("| " + " | ".join(str(v) for v in r) + " |")
        return "\n".join(lines)

    @mcp.tool()
    async def execute_write(sql: str, params: list[Any] | None = None,
                            ctx: Context = None) -> str:
        """执行写操作 SQL（INSERT / UPDATE / DELETE）。执行前会向用户确认。"""
        # Elicitation：执行前确认
        if ctx:
            result = await ctx.elicit(
                message=f"即将执行写操作：\n\n```sql\n{sql}\n```\n\n是否确认？",
                schema={
                    "type": "object",
                    "properties": {
                        "confirm": {"type": "boolean", "title": "确认执行"}
                    },
                    "required": ["confirm"],
                },
            )
            if result.action == "cancel" or not result.data.get("confirm"):
                return "操作已取消。"
        with get_conn() as conn:
            cursor = conn.execute(sql, params or [])
            conn.commit()
        return f"执行成功，影响 {cursor.rowcount} 行。"

    @mcp.resource("db://tables")
    def db_tables_resource() -> str:
        """以文本形式列出所有表（Resource 版，供上下文注入）。"""
        with get_conn() as conn:
            rows = conn.execute(
                "SELECT name FROM sqlite_master WHERE type='table'"
            ).fetchall()
        return "数据库表：\n" + "\n".join(r["name"] for r in rows)

12.4 文件系统 Tools 实现

# fs.py
import os
from pathlib import Path
from mcp.server.fastmcp import FastMCP

# 授权根目录（生产环境应从配置文件读取）
WORKSPACE = Path(os.environ.get("MCP_WORKSPACE", "/tmp/mcp-workspace")).resolve()
WORKSPACE.mkdir(parents=True, exist_ok=True)


def safe_path(rel: str) -> Path:
    """把相对路径解析为授权目录内的绝对路径，防御路径穿越。"""
    p = (WORKSPACE / rel).resolve()
    if WORKSPACE not in (p, *p.parents):
        raise PermissionError(f"路径 '{rel}' 超出授权范围。")
    return p


def register_fs(mcp: FastMCP):

    @mcp.tool()
    def list_dir(path: str = ".") -> str:
        """列出授权工作区内的目录内容。"""
        p = safe_path(path)
        if not p.is_dir():
            raise NotADirectoryError(f"'{path}' 不是目录。")
        entries = []
        for entry in sorted(p.iterdir()):
            kind = "DIR " if entry.is_dir() else "FILE"
            size = "" if entry.is_dir() else f"  {entry.stat().st_size:>10} B"
            entries.append(f"{kind}  {entry.name}{size}")
        return "\n".join(entries) if entries else "（空目录）"

    @mcp.tool()
    def read_file(path: str) -> str:
        """读取授权工作区内的文本文件（最大 1MB）。"""
        p = safe_path(path)
        if not p.is_file():
            raise FileNotFoundError(f"文件 '{path}' 不存在。")
        if p.stat().st_size > 1024 * 1024:
            raise ValueError("文件超过 1MB，请使用分块读取。")
        return p.read_text(encoding="utf-8", errors="replace")

    @mcp.tool()
    def write_file(path: str, content: str, overwrite: bool = False) -> str:
        """在授权工作区内写入文本文件。
        overwrite=False 时，若文件已存在则报错，避免意外覆盖。
        """
        p = safe_path(path)
        if p.exists() and not overwrite:
            raise FileExistsError(
                f"文件 '{path}' 已存在。如需覆盖，请将 overwrite 设为 true。"
            )
        p.parent.mkdir(parents=True, exist_ok=True)
        p.write_text(content, encoding="utf-8")
        return f"已写入 {len(content)} 个字符到 '{path}'。"

    @mcp.resource("file://{path}")
    def file_resource(path: str) -> str:
        """通过 URI 读取工作区文件（Resource 版）。"""
        return (safe_path(path)).read_text(encoding="utf-8", errors="replace")

12.5 Prompts 实现

# prompts.py
from mcp.server.fastmcp import FastMCP


def register_prompts(mcp: FastMCP):

    @mcp.prompt()
    def analyze_table(table: str) -> str:
        """生成分析某张数据库表的提示。"""
        return f"""请帮我全面分析数据库中的 `{table}` 表：
1. 调用 describe_table 查看字段定义，理解数据结构。
2. 调用 query 执行 SELECT COUNT(*) 获取数据量。
3. 调用 query 查询前 10 行，分析典型数据。
4. 根据字段类型和数据，指出可能的数据质量问题（空值、类型不一致等）。
5. 给出 2-3 个有价值的分析建议。"""

    @mcp.prompt()
    def data_to_report(table: str, output_file: str) -> str:
        """从表中提取数据并生成 Markdown 报告，保存到文件。"""
        return f"""请按以下步骤生成数据报告：
1. 调用 query 从 `{table}` 表查询最近 50 条记录。
2. 根据查询结果，生成一份包含摘要统计和关键发现的 Markdown 报告。
3. 调用 write_file，把报告写入 '{output_file}'（overwrite=true）。
4. 完成后告诉我报告的主要发现。"""

12.6 入口与启动

# server.py
from mcp.server.fastmcp import FastMCP
from db import register_db
from fs import register_fs
from prompts import register_prompts

mcp = FastMCP(
    "data-server",
    instructions=(
        "我是一个数据助手，可以查询 SQLite 数据库和读写本地文件。"
        "所有文件操作限制在授权工作区内。"
        "写操作（SQL 写入、文件覆盖）会在执行前请求确认。"
    ),
)

register_db(mcp)
register_fs(mcp)
register_prompts(mcp)

if __name__ == "__main__":
    mcp.run()
